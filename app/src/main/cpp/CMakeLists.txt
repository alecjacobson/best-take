cmake_minimum_required(VERSION 3.22)
project(besttake_jni LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(BT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(ANDROID)
  set(BT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
else()
  # Host build used by JVM unit tests of the bridge; also builds the fixture tool.
  set(BT_BUILD_TOOLS ON CACHE BOOL "" FORCE)
  if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
  endif()
endif()
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../engine ${CMAKE_CURRENT_BINARY_DIR}/engine)

add_library(besttake SHARED jni.cpp)
target_link_libraries(besttake PRIVATE bt_engine)
if(NOT ANDROID)
  # Only the JNI headers are needed; BT_JAVA_HOME is the JDK running the tests.
  if(NOT BT_JAVA_HOME OR NOT EXISTS ${BT_JAVA_HOME}/include/jni.h)
    message(FATAL_ERROR "BT_JAVA_HOME must point at a JDK with include/jni.h (got '${BT_JAVA_HOME}')")
  endif()
  if(APPLE)
    set(BT_JNI_PLATFORM darwin)
  elseif(WIN32)
    set(BT_JNI_PLATFORM win32)
  else()
    set(BT_JNI_PLATFORM linux)
  endif()
  target_include_directories(besttake PRIVATE ${BT_JAVA_HOME}/include ${BT_JAVA_HOME}/include/${BT_JNI_PLATFORM})
endif()
